# SFT→DPO on Rejected CPT-v2 (Hypothesis Probe) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Chain SFT then DPO (both MLX LoRA) on top of the rejected CPT-v2 adapter, using the already-built `fresh` 04-cpt-sft dataset, to test whether CPT-v2 provides real value once a downstream SFT/DPO stage runs on top of it — independent of CPT-v2's own Track A/B rejection. Produce extensive graphs at every stage (training curves, per-checkpoint functional pass-rate, DPO preference accuracy/margin) and a final 3-way comparison (CPT-v2-base vs +SFT vs +SFT+DPO).

**Architecture:** New self-contained scaffold at `model-experiments/04-cpt-sft/sft_cptv2_probe/`. Fresh fresh 85/15 split of `sft_train.jsonl` (9608 rows) and `dpo_train.jsonl` (826 rows) — ignores the frozen 18-id pilot holdout (that stays reserved for the formal 3-arm protocol). SFT LoRA resumes directly from the CPT-v2 adapter (`--resume-adapter-file`, no fuse needed, structural match required: rank16/num_layers16). SFT adapter is then fused into the base model, and a fresh rank16/layers16 DPO LoRA trains on top of that fused model via the third-party `mlx-lm-lora` trainer (mlx_lm has no native DPO). A generalized functional eval (`eval_functional.jac`, extending the existing `eval_probe.jac` pattern) measures jac-compiles-and-runs rate by category/gate_class at CPT-v2-base, every SFT checkpoint, and DPO-final. DPO's own trainer already emits accuracy/margin metrics per interval — a new log parser plots those directly, no custom preference-scoring script needed.

**Tech Stack:** mlx-lm 0.31.3, mlx-lm-lora 2.1.0, jaclang (`jac run`/`jac check`), matplotlib (Agg backend), Python 3.14 venv at repo root `.venv`.

## Global Constraints

- Base model: `models/qwen-q4` (Qwen/Qwen3-Coder-30B-A3B-Instruct, Q4 MLX quant) — the exact base CPT-v2 was trained against. Do not swap base models mid-plan.
- Any new SFT adapter that resumes from the CPT-v2 adapter MUST use `num_layers: 16`, `lora_parameters.rank: 16` — shape-mismatch on `--resume-adapter-file` silently corrupts or fails to load otherwise.
- Disk: ExtremePro drive has a known flaky-USB history (see memory `reference-repo-migration`) — never run destructive cleanup on `adapters/`/`results/` without confirming with the user first; treat all trained checkpoints as precious per project convention (never delete models).
- No `LIVE_EVAL` (no concurrent second-model-load holdout eval while training is running) — confirmed OOM/deadlock risk on 48GB for a 30B model (see `run_probe.sh` header). Functional eval always runs sequentially, one checkpoint at a time, after its training stage fully completes.
- All new files live under `model-experiments/04-cpt-sft/sft_cptv2_probe/` — do not modify `jacgen2/`, the frozen holdout files, or `dataset/fresh/releases/*.jsonl` in place.
- `run_jac(code, timeout=30)` (from `model-experiments/01-sft-dpo/sft_dpo/jacgen/writer.jac`) is the only code-execution gate to reuse — returns `(returncode, stdout, stderr)`, isolates `root`/db state per call via a scoped tempdir `cwd`.

- [ ] **Step 1: Write `split_probe.jac`**

```jac
"""Fresh 85/15 split for the CPT-v2-SFT/DPO hypothesis probe.

Deliberately ignores the frozen 18-id pilot holdout in dataset/shared/holdouts/
(that stays reserved for the formal 3-arm protocol) -- this is a separate,
one-off quick test, sized properly against the real 9608/826-row datasets.

Deterministic shuffle (seed 42, matches the project's existing lora.yaml
convention) then a straight 85/15 slice, not a stride split -- at ~1441/124
holdout rows a stride split and a shuffled slice are statistically
equivalent, and a slice is simpler to reason about.

Run from the repo root:
    jac run model-experiments/04-cpt-sft/sft_cptv2_probe/jacgen/split_probe.jac
"""

import json;
import random;
import from pathlib { Path }
import from typing { Any }

glob SFT_SRC: str = "model-experiments/04-cpt-sft/dataset/fresh/releases/sft_train.jsonl";
glob DPO_SRC: str = "model-experiments/04-cpt-sft/dataset/fresh/releases/dpo_train.jsonl";
glob OUT_ROOT: str = "model-experiments/04-cpt-sft/sft_cptv2_probe/dataset";
glob SEED: int = 42;
glob VALID_FRAC: float = 0.15;
glob TOKENIZER_MODEL: str = "models/qwen-q4";


def read_lines(path: str) -> list[dict] {
    rows: list[dict] = [];
    p = Path(path);
    for line in p.read_text().split("\n") {
        if line.strip() == "" { continue; }
        rows.append(dict(json.loads(line)));
    }
    return rows;
}


def write_jsonl(path: str, rows: list[dict]) {
    with open(path, "w") as f {
        for r in rows { f.write(json.dumps(r) + "\n"); }
    }
}


def split_rows(rows: list[dict]) -> tuple[list[dict], list[dict]] {
    idx: list[int] = list(range(len(rows)));
    rng = random.Random(SEED);
    rng.shuffle(idx);
    n_valid: int = int(round(len(rows) * VALID_FRAC));
    valid_idx: set = set(idx[:n_valid]);
    train: list[dict] = [];
    valid: list[dict] = [];
    for (i, r) in enumerate(rows) {
        if i in valid_idx { valid.append(r); } else { train.append(r); }
    }
    return (train, valid);
}


def token_stats(texts: list[str], tok: Any) -> dict {
    counts: list[int] = [];
    for t in texts { counts.append(len(tok.encode(t))); }
    counts.sort();
    n: int = len(counts);
    def pct(p: float) -> int {
        if n == 0 { return 0; }
        i: int = int(p * float(n - 1));
        return counts[i];
    }
    mx: int = 0;
    if n > 0 { mx = counts[n - 1]; }
    return {"p50": pct(0.50), "p95": pct(0.95), "p99": pct(0.99), "max": mx, "n": n};
}


with entry {
    Path(OUT_ROOT + "/sft").mkdir(parents=True, exist_ok=True);
    Path(OUT_ROOT + "/dpo").mkdir(parents=True, exist_ok=True);

    # --- SFT split ---
    sft_rows: list[dict] = read_lines(SFT_SRC);
    (sft_train, sft_valid) = split_rows(sft_rows);
    sft_train_stripped: list[dict] = [{"messages": r["messages"]} for r in sft_train];
    write_jsonl(OUT_ROOT + "/sft/train.jsonl", sft_train_stripped);
    write_jsonl(OUT_ROOT + "/sft/valid.jsonl", sft_valid);  # full rows -- doubles as functional-eval holdout
    print("SFT train: " + str(len(sft_train)) + "  valid(holdout): " + str(len(sft_valid)));

    # --- DPO split ---
    dpo_rows: list[dict] = read_lines(DPO_SRC);
    (dpo_train, dpo_valid) = split_rows(dpo_rows);
    dpo_train_stripped: list[dict] = [{"prompt": str(r["prompt"]), "chosen": str(r["chosen"]),
                                        "rejected": str(r["rejected"])} for r in dpo_train];
    dpo_valid_stripped: list[dict] = [{"prompt": str(r["prompt"]), "chosen": str(r["chosen"]),
                                        "rejected": str(r["rejected"])} for r in dpo_valid];
    write_jsonl(OUT_ROOT + "/dpo/train.jsonl", dpo_train_stripped);
    write_jsonl(OUT_ROOT + "/dpo/valid.jsonl", dpo_valid_stripped);
    print("DPO train: " + str(len(dpo_train)) + "  valid: " + str(len(dpo_valid)));

    # --- token-length stats (sizes max_seq_length in Task 2/6's configs) ---
    import mlx_lm;   # lazy import, same convention as eval_probe.jac
    (_model, tok) = mlx_lm.load(TOKENIZER_MODEL);  # jac:ignore[E1032]

    sft_texts: list[str] = [];
    for r in sft_rows {
        joined: str = "";
        for m in r["messages"] { joined += str(m["content"]) + "\n"; }
        sft_texts.append(joined);
    }
    dpo_texts: list[str] = [];
    for r in dpo_rows {
        dpo_texts.append(str(r["prompt"]) + "\n" + str(r["chosen"]));
        dpo_texts.append(str(r["prompt"]) + "\n" + str(r["rejected"]));
    }

    stats: dict = {"sft": token_stats(sft_texts, tok), "dpo": token_stats(dpo_texts, tok)};
    with open(OUT_ROOT + "/dataset_stats.json", "w") as f { f.write(json.dumps(stats, indent=2)); }
    print("token stats -> " + OUT_ROOT + "/dataset_stats.json");
    print(json.dumps(stats, indent=2));
}
```

`iters: 4100` = ~1 epoch at `batch_size 2` over the 8167-row train split (`4100 * 2 / 8167 ≈ 1.00`). - [ ] **Step 2: Write `run_sft.sh`**

```bash
#!/usr/bin/env bash
# SFT-on-CPT-v2 probe runner. Mirrors 01-sft-dpo/sft_dpo/run_probe.sh's structure:
# resumable, dry-run-first, live log-tail dashboard (NOT a second model load --
# LIVE_EVAL stays off, confirmed OOM risk on 48GB for a 30B model).
set -euo pipefail

if [ -z "${CAFFEINATED:-}" ] && command -v caffeinate >/dev/null 2>&1; then
  exec caffeinate -dimsu env CAFFEINATED=1 "$0" "$@"
fi

SELF_DIR="$(cd "$(dirname "$0")" && pwd)"
cd "$(cd "$SELF_DIR/../../.." && pwd)"   # repo root
[ -d ".venv/bin" ] && export PATH="$PWD/.venv/bin:$PATH"

CFG="model-experiments/04-cpt-sft/sft_cptv2_probe/configs/sft.yaml"
ADAPTER="model-experiments/04-cpt-sft/sft_cptv2_probe/adapters/sft-on-cptv2"
RDIR="model-experiments/04-cpt-sft/sft_cptv2_probe/results/sft"
TRAIN_LOG="$RDIR/train.log"
DRY_ITERS="${DRY_ITERS:-30}"
EVAL_EVERY="${EVAL_EVERY:-60}"

need() { command -v "$1" >/dev/null 2>&1 || { echo "MISSING: $1"; exit 1; }; }
need jac "pip install jaclang"; need mlx_lm.lora "pip install mlx-lm"
for f in model-experiments/04-cpt-sft/sft_cptv2_probe/dataset/sft/train.jsonl \
         model-experiments/04-cpt-sft/sft_cptv2_probe/dataset/sft/valid.jsonl \
         model-experiments/03-cpt-only/adapters/cpt-v2/adapters.safetensors "$CFG"; do
  [ -f "$f" ] || { echo "MISSING: $f"; exit 1; }
done

mkdir -p "$RDIR" "$ADAPTER"
TRAIN_PID=""
cleanup() { [ -n "$TRAIN_PID" ] && kill "$TRAIN_PID" 2>/dev/null || true; }
trap cleanup EXIT INT TERM
done_mark() { touch "$RDIR/.$1.done"; }
is_done() { [ -f "$RDIR/.$1.done" ]; }

# --- dry-run (only on a truly fresh start) ---
LATEST_CKPT="$(ls "$ADAPTER"/*_adapters.safetensors 2>/dev/null | sort -V | tail -1 || true)"
if [ -z "$LATEST_CKPT" ] && ! is_done dry && [ "${SKIP_DRY:-0}" != "1" ]; then
  echo ">>> dry-run (${DRY_ITERS} iters) -- bail check"
  mlx_lm.lora --config "$CFG" --iters "$DRY_ITERS" \
    --adapter-path "model-experiments/04-cpt-sft/sft_cptv2_probe/adapters/dry" 2>&1 | tail -25
  echo ">>> dry-run done -- Ctrl-C within 8s to abort"; sleep 8
  done_mark dry
fi

TOTAL_ITERS="$(grep -E '^iters:' "$CFG" | grep -oE '[0-9]+' | head -1)"
DONE_STEPS=0
if [ -n "$LATEST_CKPT" ]; then
  DONE_STEPS="$(basename "$LATEST_CKPT" | grep -oE '^[0-9]+' | sed 's/^0*//')"; DONE_STEPS="${DONE_STEPS:-0}"
  echo ">>> resuming from step ${DONE_STEPS}"
fi
REMAIN=$(( TOTAL_ITERS - DONE_STEPS ))
ADAPTER_FILE="$ADAPTER/adapters.safetensors"

if { is_done train || [ "$REMAIN" -le 0 ]; } && [ -f "$ADAPTER_FILE" ]; then
  echo ">>> training: already complete (${DONE_STEPS}/${TOTAL_ITERS})"
else
  echo ">>> training ${REMAIN} more iters (from ${DONE_STEPS}/${TOTAL_ITERS})"
  : > "$TRAIN_LOG"
  if [ -n "$LATEST_CKPT" ]; then
    mlx_lm.lora --config "$CFG" --adapter-path "$ADAPTER" --iters "$REMAIN" \
      --resume-adapter-file "$LATEST_CKPT" > "$TRAIN_LOG" 2>&1 &
  else
    mlx_lm.lora --config "$CFG" --adapter-path "$ADAPTER" --iters "$REMAIN" \
      > "$TRAIN_LOG" 2>&1 &
  fi
  TRAIN_PID=$!
  while kill -0 "$TRAIN_PID" 2>/dev/null; do
    JAC_TRAIN_LOG="$TRAIN_LOG" JAC_METRICS="/dev/null" JAC_PLOT_DIR="$RDIR" \
      jac run model-experiments/01-sft-dpo/sft_dpo/jacgen/plot_metrics.jac >/dev/null 2>&1 || true
    sleep "$EVAL_EVERY"
  done
  RC=0; wait "$TRAIN_PID" || RC=$?
  TRAIN_PID=""
  if [ "$RC" -ne 0 ] || [ ! -f "$ADAPTER_FILE" ]; then
    echo "!!! training stopped early (exit $RC). Re-run this script to resume."
    tail -20 "$TRAIN_LOG"; exit 1
  fi
  done_mark train
fi
echo "=== SFT training done. Next: Task 3 (functional eval sweep) ==="
```

### Task 4: Generalized functional eval script (`eval_functional.jac`)

**Interfaces:**

- [ ] **Step 1: Write `eval_functional.jac`**

```jac
"""Generalized functional-eval harness for the CPT-v2-SFT/DPO probe.

Extends eval_probe.jac's mechanism (load model once, generate, extract fenced
jac, execute via run_jac) to the full 7-category sft_train.jsonl schema instead
of the narrow func_name/test_cases conversion-only shape. There is no
per-example expected-output here (most categories never had test_cases), so
this measures ONLY "runs (compiles+executes)" -- not behavioral correctness.
Rows are routed by their own `gate_class` field: "prose_lexical" rows (mostly
explanation/documentation) are generated but NOT code-graded (no code output
expected) -- reported separately, never silently dropped from the totals.

    JAC_EVAL_MODE=mlx JAC_EVAL_MODEL=models/qwen-q4 [JAC_EVAL_ADAPTER=path] \
      JAC_HOLDOUT=model-experiments/04-cpt-sft/sft_cptv2_probe/dataset/sft/valid.jsonl \
      [JAC_EVAL_LIMIT=100] [JAC_EVAL_METRICS_OUT=path] [JAC_EVAL_STEP=0] \
      jac run model-experiments/04-cpt-sft/sft_cptv2_probe/jacgen/eval_functional.jac
"""

import json;
import os;
import from writer { run_jac }
import from pathlib { Path }
import from typing { Any }

glob HOLDOUT: str = "model-experiments/04-cpt-sft/sft_cptv2_probe/dataset/sft/valid.jsonl";
glob CODE_GATES: set = {"behavioral", "compile_project", "compile_only"};


def load_model(mp: str, adapter: str) -> tuple {
    import mlx_lm;
    if adapter != "" { return mlx_lm.load(mp, adapter_path=adapter); }  # jac:ignore[E1032]
    return mlx_lm.load(mp);  # jac:ignore[E1032]
}


def gen_mlx(model: Any, tok: Any, prompt: str) -> tuple {
    import mlx_lm;
    msgs: list = [{"role": "user", "content": prompt}];
    ptoks = tok.apply_chat_template(msgs, add_generation_prompt=True);
    text: str = "";
    gtok: int = 0;
    for resp in mlx_lm.stream_generate(model, tok, ptoks, max_tokens=768) {  # jac:ignore[E1053,E1032]
        text += str(resp.text);
        gtok = int(resp.generation_tokens);
    }
    return (text, gtok);
}


def extract_jac(output: str) -> str {
    body: str = output;
    if "```jac" in output { body = output.split("```jac", 1)[1].split("```", 1)[0]; }
    elif "```" in output { body = output.split("```", 1)[1].rsplit("```", 1)[0]; }
    return body.strip();
}


with entry {
    p = Path(os.environ.get("JAC_HOLDOUT", HOLDOUT));

    (model, tok) = load_model(mp, adapter);

    buckets: dict = {};   # (category, gate_class) -> {"total": n, "runs": n}

        category: str = str(rec["category"]);
        gate_class: str = str(rec["gate_class"]);
        user_msg: str = str(rec["messages"][0]["content"]);
        total += 1;

            prose_total += 1;
            continue;   # generated but not code-graded -- reported separately, not silently dropped
        }

        jac_code: str = extract_jac(out);

        (code, _sout, _serr) = run_jac(jac_code);
        if code == 0 { buckets[key]["runs"] += 1; }
    }

    overall_total: int = 0;
    overall_runs: int = 0;
    print("=== functional eval on " + str(total) + " holdout rows (" + str(prose_total) + " prose/ungraded) ===");
    metrics_out: str = os.environ.get("JAC_EVAL_METRICS_OUT", "");
    step: int = int(os.environ.get("JAC_EVAL_STEP", "0"));

        (category, gate_class) = key;
        pct: int = 0;
        if v["total"] > 0 { pct = v["runs"] * 100 // v["total"]; }
        print("  " + category + " / " + gate_class + ": " + str(pct) + "% (" + str(v["runs"]) + "/" + str(v["total"]) + ")");

        overall_runs += v["runs"];
        rows_out.append({"step": step, "category": category, "gate_class": gate_class,
                          "total": v["total"], "runs": v["runs"], "runs_pct": pct});
    }
    overall_pct: int = 0;
    if overall_total > 0 { overall_pct = overall_runs * 100 // overall_total; }
    print("OVERALL runs: " + str(overall_pct) + "% (" + str(overall_runs) + "/" + str(overall_total) + ")");
    rows_out.append({"step": step, "category": "__overall__", "gate_class": "__all__",
                      "total": overall_total, "runs": overall_runs, "runs_pct": overall_pct});

    if metrics_out != "" {
        with open(metrics_out, "a") as f {
            for r in rows_out { f.write(json.dumps(r) + "\n"); }
        }
    }
}

The only gap is that its log format differs from `plot_metrics.jac`'s regexes (`loss`/`lr`/`it/s`/`tok/s` lowercase, plus `chosen_r`/`rejected_r`/`acc`/`margin` — none of which `plot_metrics.jac` looks for), so a small dedicated parser is needed instead of reusing that script unmodified.

- [ ] **Step 1: Write `plot_dpo.jac`**

```jac
"""DPO-stage graphs, parsing mlx_lm_lora's own train-log format (confirmed
DIFFERENT from mlx_lm.lora's -- lowercase `loss`/`lr`/`it/s`/`tok/s`, plus
native `acc`/`margin`/`chosen_r`/`rejected_r` fields plot_metrics.jac has no
regex for). acc/margin ARE the preference-accuracy metric this probe needs --
the trainer computes them from its own frozen reference-model comparison, no
separate scoring script required.

    JAC_DPO_LOG=path JAC_PLOT_DIR=path jac run .../plot_dpo.jac
"""

import re;
import os;
import matplotlib;
import from matplotlib { pyplot }
import from pathlib { Path }

def _num(line: str, pattern: str) -> float {
    m = re.search(pattern, line);
    if m is None { return -1.0; }
    return float(str(m.group(1)));
}

def parse(path: str) -> dict {
    loss: list = []; acc: list = []; margin: list = [];
    chosen_r: list = []; rejected_r: list = [];
    val_loss: list = []; val_acc: list = []; val_margin: list = [];
    p = Path(path);
    if not p.exists() { return {"loss": loss, "acc": acc, "margin": margin,
                                  "chosen_r": chosen_r, "rejected_r": rejected_r,
                                  "val_loss": val_loss, "val_acc": val_acc, "val_margin": val_margin}; }
    for line in p.read_text().split("\n") {
        m = re.search("Iter (\\d+)", line);
        if m is None { continue; }
        it: int = int(str(m.group(1)));
        is_val: bool = "Val loss" in line;
        if is_val {
            vl: float = _num(line, "Val loss ([0-9.]+)");
            if vl >= 0.0 { val_loss.append([it, vl]); }
            va: float = _num(line, "Val accuracy ([0-9.]+)");
            if va >= 0.0 { val_acc.append([it, va]); }
            vm: float = _num(line, "Val margin ([0-9.-]+)");
            if vm > -999.0 { val_margin.append([it, vm]); }
        } else {
            tl: float = _num(line, "loss ([0-9.]+)");
            if tl >= 0.0 { loss.append([it, tl]); }
            ac: float = _num(line, "acc ([0-9.]+)");
            if ac >= 0.0 { acc.append([it, ac]); }
            mg: float = _num(line, "margin ([0-9.-]+)");
            if mg > -999.0 { margin.append([it, mg]); }
            cr: float = _num(line, "chosen_r ([0-9.-]+)");
            if cr > -999.0 { chosen_r.append([it, cr]); }
            rr: float = _num(line, "rejected_r ([0-9.-]+)");
            if rr > -999.0 { rejected_r.append([it, rr]); }
        }
    }
    return {"loss": loss, "acc": acc, "margin": margin, "chosen_r": chosen_r,
            "rejected_r": rejected_r, "val_loss": val_loss, "val_acc": val_acc, "val_margin": val_margin};
}

def plot_pair(train_pts: list, val_pts: list, title: str, ylabel: str, outpath: str) {
    if len(train_pts) == 0 and len(val_pts) == 0 { return; }
    pyplot.figure();  # jac:ignore[E1051]
    if len(train_pts) > 0 {
        pyplot.plot([x[0] for x in train_pts], [x[1] for x in train_pts], marker="o", label="train");  # jac:ignore[E1051,E1052]

- [ ] **Step 1: Write `make_dashboard.jac`**

```jac
"""Final 3-way comparison: CPT-v2-base vs +SFT vs +SFT+DPO.

Reads the three OVERALL/per-category runs_pct numbers already computed by
eval_functional.jac at each stage's endpoint (base.txt-adjacent metrics rows,
not re-parsed from text -- pulls from metrics_functional.jsonl's step=0 (base)
and step=<final SFT iters> rows, plus a fresh metrics row for the DPO-final
point since Task 8 Step 5 wrote it to /dev/null -- re-run that eval with a
real JAC_EVAL_METRICS_OUT first if you want this script to read it instead of
hand-transcribing the OVERALL % from final.txt).
"""

import json;
import os;
import matplotlib;
import from matplotlib { pyplot }
import from pathlib { Path }

glob SFT_METRICS: str = "model-experiments/04-cpt-sft/sft_cptv2_probe/results/sft/metrics_functional.jsonl";
glob DPO_METRICS: str = "model-experiments/04-cpt-sft/sft_cptv2_probe/results/dpo/metrics_functional.jsonl";
glob OUT_DIR: str = "model-experiments/04-cpt-sft/sft_cptv2_probe/results/images";


def load_rows(path: str) -> list[dict] {
    rows: list[dict] = [];
    p = Path(path);
    if not p.exists() { return rows; }
    for line in p.read_text().split("\n") {
        if line.strip().startswith("{") { rows.append(dict(json.loads(line))); }
    }
    return rows;
}


with entry {
    matplotlib.use("Agg");
    Path(OUT_DIR).mkdir(parents=True, exist_ok=True);

    sft_rows: list[dict] = load_rows(SFT_METRICS);
    dpo_rows: list[dict] = load_rows(DPO_METRICS);

    base_overall: dict = {};
    sft_final_overall: dict = {};
    max_sft_step: int = 0;
    for r in sft_rows {
        if r["category"] != "__overall__" { continue; }
        if r["step"] == 0 { base_overall = r; }
        if r["step"] >= max_sft_step { max_sft_step = r["step"]; sft_final_overall = r; }
    }
    dpo_final_overall: dict = {};
    max_dpo_step: int = -1;
    for r in dpo_rows {
        if r["category"] != "__overall__" { continue; }
        if r["step"] >= max_dpo_step { max_dpo_step = r["step"]; dpo_final_overall = r; }
    }

    labels: list[str] = ["CPT-v2 base", "+SFT"];
    values: list[float] = [float(base_overall.get("runs_pct", 0)), float(sft_final_overall.get("runs_pct", 0))];
    if len(dpo_final_overall) > 0 {
        labels.append("+SFT+DPO");
        values.append(float(dpo_final_overall.get("runs_pct", 0)));
    } else {
        print("NOTE: no DPO metrics_functional.jsonl found -- re-run Task 8 Step 5 with a real "
              + "JAC_EVAL_METRICS_OUT path (not /dev/null) to include the +SFT+DPO bar.");
    }

    pyplot.figure(figsize=(7, 5));  # jac:ignore[E1051]
    pyplot.bar(labels, values, color=["#888888", "#4C72B0", "#55A868"][:len(labels)]);  # jac:ignore[E1051]
    pyplot.title("Functional pass rate (runs %) -- hypothesis probe");  # jac:ignore[E1051]
    pyplot.ylabel("runs %");  # jac:ignore[E1051]
    pyplot.ylim(0, 100);  # jac:ignore[E1051]
    for (i, v) in enumerate(values) { pyplot.text(i, v + 1, str(round(v, 1)) + "%", ha="center"); }  # jac:ignore[E1051]
    pyplot.grid(True, axis="y");  # jac:ignore[E1051]
    pyplot.savefig(OUT_DIR + "/comparison_overall.png");  # jac:ignore[E1051]
    pyplot.close();  # jac:ignore[E1051]
    print("wrote " + OUT_DIR + "/comparison_overall.png");
    print(json.dumps({"base": base_overall, "sft_final": sft_final_overall, "dpo_final": dpo_final_overall}, indent=2));
}
```
